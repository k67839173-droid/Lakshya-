```tsx name=components/FragranceFinder.tsx
import { useState } from "react";
import { PRODUCTS } from "../data/products";

const OPTIONS = ["Romantic","Calm","Bold","Mysterious","Fresh","Warm"];

export default function FragranceFinder(){
  const [selected, setSelected] = useState<string | null>(null);
  const recommend = (mood: string) => {
    // Simple mapping: Romantic -> GUL, Mysterious -> NOOR, Bold -> DHUN, Warm -> GUL, Fresh -> NOOR
    if (mood === "Romantic") return PRODUCTS.find(p => p.slug === "gul");
    if (mood === "Mysterious") return PRODUCTS.find(p => p.slug === "noor");
    if (mood === "Bold") return PRODUCTS.find(p => p.slug === "dhun");
    if (mood === "Warm") return PRODUCTS.find(p => p.slug === "gul");
    if (mood === "Fresh") return PRODUCTS.find(p => p.slug === "noor");
    return PRODUCTS[0];
  };
  const result = selected ? recommend(selected) : null;

  return (
    <section id="discover" className="section-space container">
      <div className="text-center mb-8">
        <h3 className="text-2xl h2">FIND YOUR TRAMBH</h3>
        <p className="text-[#CFC6B4] mt-2">What mood are you feeling?</p>
      </div>

      <div className="flex flex-col md:flex-row items-start md:items-center gap-6 justify-center">
        <div className="grid grid-cols-3 md:grid-cols-6 gap-4">
          {OPTIONS.map(opt => (
            <button key={opt} onClick={() => setSelected(opt)} className={`px-4 py-2 text-sm rounded uppercase tracking-widest ${selected === opt ? 'bg-trambh-gold text-black' : 'bg-transparent border border-[#222] text-[#D9D1BC]'}`}>
              {opt}
            </button>
          ))}
        </div>

        <div className="mt-4 md:mt-0">
          {result ? (
            <div className="card-lux p-5 rounded-md">
              <div className="text-sm text-[#CFC6B4]">Recommended</div>
              <div className="mt-2 font-semibold" style={{ fontFamily: "'Playfair Display', serif" }}>{result.name}</div>
              <div className="text-sm mt-1 text-[#CFC6B4]">{result.mood}</div>
              <a href={`/products/${result.slug}`} className="mt-4 inline-block px-4 py-2 border border-trambh-gold text-sm text-trambh-gold">VIEW FRAGRANCE</a>
            </div>
          ) : (
            <div className="text-[#CFC6B4]">Select a mood to see a recommendation.</div>
          )}
        </div>
      </div>
    </section>
  );
}
